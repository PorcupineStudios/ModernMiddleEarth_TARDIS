# The colony management screen (UI core), for the "UI screenshots" workflow: a colony with dwarves, hobbits and men,
# then the People tab, its filter box, Overview, Buildings and Members, and one citizen's page. Click spots are GUI
# units for a 1280x720 window at GUI scale 3 (427 x 240): see UiScreen's layout.
wait 200
/time set 6000
/weather clear
/gamerule doMobSpawning false
/lotr colony found "Minas Test"
/lotr npc spawn lotr_port:blue_dwarf
/lotr npc spawn lotr_port:blue_dwarf
/lotr npc spawn lotr_port:blue_dwarf
/lotr npc spawn lotr_port:blue_dwarf
/lotr npc spawn lotr_port:blue_dwarf

/lotr npc spawn lotr_port:gondor_man
/lotr npc spawn lotr_port:gondor_man
